    <script>

        function profitCalculator() {

            return {

                platform: 'EBAY',

                sellerType: 'PRIVATE',

                vatScheme: 'NONE',

                currency: 'GBP',

                

                soldPrice: '',

                costPrice: '',

                postage: '',

                

                platformFeePct: 12.9,

                flatRatePct: 7.5,


                init() {

                    // Update default fee percentage visually for business users when platform changes

                    this.$watch('platform', (val) => {

                        if (val === 'EBAY') this.platformFeePct = 12.9;

                        if (val === 'SHOPIFY') this.platformFeePct = 2.0;

                        if (val === 'ETSY') this.platformFeePct = 10.5; // Roughly 6.5 + 4

                    });

                },


                get currencySymbol() {

                    return this.currency === 'GBP' ? '£' : '$';

                },


                get buyerFee() {

                    const s = parseFloat(this.soldPrice) || 0;

                    const fixed = this.currency === 'GBP' ? 0.70 : 0.90;

                    

                    if (this.platform === 'VINTED') return (s * 0.05) + fixed;

                    if (this.platform === 'DEPOP') return (s * 0.05); // 5% marketplace fee

                    return 0; // eBay/Etsy/Shopify

                },


                get results() {

                    const s = parseFloat(this.soldPrice) || 0;

                    const c = parseFloat(this.costPrice) || 0;

                    const p = parseFloat(this.postage) || 0;

                    const pfPct = parseFloat(this.platformFeePct) || 0;

                    const frPct = parseFloat(this.flatRatePct) || 0;


                    let fees = 0;

                    let vat = 0;

                    const totalRevenue = s + p; // Most fees charge on postage too


                    // --- Platform Fee Logic ---

                    if (this.platform === 'VINTED' || this.platform === 'DEPOP') {

                        fees = 0; 

                    } 

                    else if (this.platform === 'ETSY') {

                        // 20p listing + (6.5% trans + 4% payment) + 20p payment fixed

                        // Note: Payment fee applies to Tax too, but simplistic here.

                        // Approx: 10.5% + 40p

                        fees = (totalRevenue * 0.105) + 0.40;

                    } 

                    else if (this.platform === 'SHOPIFY') {

                        // 2% + 25p

                        fees = (totalRevenue * 0.02) + 0.25;

                    } 

                    else {

                        // EBAY UK

                        if (this.sellerType === 'PRIVATE') {

                            fees = 0;

                        } else {

                            // Business: ~12.9% + 30p

                            // We use the input percentage for flexibility + fixed 30p

                            fees = (totalRevenue * (pfPct / 100)) + 0.30;

                        }

                    }


                    // --- VAT Logic (Business Only) ---

                    if (this.sellerType === 'BUSINESS') {

                        if (this.vatScheme === 'STANDARD') {

                            // Standard 20%

                            const outputVat = s - (s / 1.2);

                            const inputVatItems = c - (c / 1.2);

                            // Fees usually have VAT on top or included. 

                            // If calculated above as "Gross Fee", we extract VAT.

                            const inputVatFees = fees - (fees / 1.2);

                            

                            vat = outputVat - inputVatItems - inputVatFees;

                        } else if (this.vatScheme === 'FLAT') {

                            // Flat Rate on Gross Turnover

                            vat = s * (frPct / 100);

                        }

                    }


                    const profit = s - c - p - fees - vat;

                    const margin = s > 0 ? (profit / s) * 100 : 0;


                    return {

                        fees,

                        vat,

                        profit,

                        margin

                    };

                },


                formatMoney(value) {

                    return new Intl.NumberFormat(this.currency === 'GBP' ? 'en-GB' : 'en-US', {

                        style: 'currency',

                        currency: this.currency,

                        minimumFractionDigits: 2

                    }).format(value);

                },


                reset() {

                    this.soldPrice = '';

                    this.costPrice = '';

                    this.postage = '';

                }

            }

        }
